% Unicode-support via fontspec (lualatex)
\usepackage{fontspec}
% DejaVu when installed, otherwise TeX Gyre fonts that ship with TeX Live (issue #51).
\IfFontExistsTF{DejaVu Serif}
  {\setmainfont{DejaVu Serif}}
  {\setmainfont{TeX Gyre Termes}}
\IfFontExistsTF{DejaVu Sans}
  {\setsansfont{DejaVu Sans}}
  {\setsansfont{TeX Gyre Heros}}
\IfFontExistsTF{DejaVu Sans Mono}
  {\setmonofont{DejaVu Sans Mono}}
  {\setmonofont{TeX Gyre Cursor}}
\usepackage{newunicodechar}
% The check mark needs a font with that glyph; the TeX Gyre fonts lack it.
\IfFontExistsTF{DejaVu Sans}
  {\newfontfamily\symbolfont{DejaVu Sans}}
  {\IfFontExistsTF{Menlo}
    {\newfontfamily\symbolfont{Menlo}}
    {\IfFontExistsTF{Noto Sans Symbols2}
      {\newfontfamily\symbolfont{Noto Sans Symbols2}}
      {\IfFontExistsTF{Symbola}
        {\newfontfamily\symbolfont{Symbola}}
        {\newfontfamily\symbolfont{TeX Gyre Heros}}}}}
\newunicodechar{✔}{{\symbolfont ✔}}

% Verplicht apart voorblad voor reader-PDF's.
% Pandoc's standaard LaTeX-template zet \maketitle direct voor de inhoudsopgave.
% Zonder herdefinitie kunnen titel en TOC op dezelfde pagina terechtkomen.
\makeatletter
% Een coverImage uit de frontmatter definieert \bsocoverimage, en readerCoverLogo
% uit de config definieert \bsocoverlogo (zie reader-pdf-converter.ts).
\renewcommand{\maketitle}{%
  \begin{titlepage}
    \centering
    \ifdefined\bsocoverimage\vspace*{0.08\textheight}\else\vspace*{0.18\textheight}\fi
    {\Huge\bfseries \@title\par}
    \vspace{1.5em}
    {\Large \@author\par}
    \ifdefined\bsocoverimage
      \vspace{2.5em}
      \includegraphics[width=0.8\textwidth,height=0.45\textheight,keepaspectratio]{\bsocoverimage}\par
    \fi
    \vfill
    \ifdefined\bsocoverlogo
      \includegraphics[width=4cm,height=3cm,keepaspectratio]{\bsocoverlogo}\par
      \vspace{1.5em}
    \fi
    {\large \ifdefined\bsocoverdates\bsocoverdates\else\@date\fi\par}
  \end{titlepage}
}
\makeatother
